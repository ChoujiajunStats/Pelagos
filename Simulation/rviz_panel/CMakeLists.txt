cmake_minimum_required(VERSION 3.16)
project(pelagos_rviz)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(pelagos_mission_tools SHARED src/mission_tools.cpp)
target_include_directories(pelagos_mission_tools PUBLIC include)
ament_target_dependencies(pelagos_mission_tools rviz_common pluginlib rclcpp std_msgs)
target_link_libraries(pelagos_mission_tools Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS pelagos_mission_tools LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_package()
