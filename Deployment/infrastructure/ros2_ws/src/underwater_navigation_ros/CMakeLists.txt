cmake_minimum_required(VERSION 3.18)
project(underwater_navigation_ros LANGUAGES CXX)
get_filename_component(DAPENG_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
find_package(ament_cmake REQUIRED)
foreach(dependency rclcpp rcl_interfaces geometry_msgs nav_msgs sensor_msgs std_msgs std_srvs visualization_msgs diagnostic_msgs tf2 tf2_ros tf2_geometry_msgs)
  find_package(${dependency} REQUIRED)
endforeach()
add_subdirectory("${DAPENG_ROOT}/Deployment/contexts/navigation/core" navigation_core)
add_executable(wavemap_navigation_node src/navigation_node.cpp src/native_visualizer.cpp)
target_compile_features(wavemap_navigation_node PRIVATE cxx_std_17)
target_compile_options(wavemap_navigation_node PRIVATE -Wall -Wextra -Wpedantic -g1)
target_link_libraries(wavemap_navigation_node dapeng_navigation_core)
ament_target_dependencies(wavemap_navigation_node rclcpp rcl_interfaces geometry_msgs nav_msgs sensor_msgs std_msgs std_srvs visualization_msgs diagnostic_msgs tf2 tf2_ros tf2_geometry_msgs)
install(TARGETS wavemap_navigation_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY "${DAPENG_ROOT}/Deployment/contexts/navigation/config" DESTINATION share/${PROJECT_NAME})
install(PROGRAMS "${DAPENG_ROOT}/Deployment/interfaces/visualization/adapters/robot_model.py" DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY meshes DESTINATION share/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES "${DAPENG_ROOT}/Deployment/interfaces/visualization/rviz/robot_incremental_mapping.rviz" DESTINATION share/${PROJECT_NAME}/rviz)
ament_package()
