cmake_minimum_required(VERSION 3.18)
project(dapeng_navigation_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(yaml-cpp REQUIRED)
if(NOT TARGET wavemap_core)
  if(NOT WAVEMAP_SOURCE_DIR)
    get_filename_component(WAVEMAP_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../TP/Mapping/wavemap" ABSOLUTE)
  endif()
  set(GENERATE_WAVEMAP_INSTALL_RULES OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  add_subdirectory("${WAVEMAP_SOURCE_DIR}/library/cpp" "${CMAKE_CURRENT_BINARY_DIR}/wavemap")
endif()
if(NOT WAVESTAR_SOURCE_DIR)
  get_filename_component(WAVESTAR_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../TP/Planner/wavestar" ABSOLUTE)
endif()
# Build the requested upstream planners unchanged. OMPL is not needed by these
# algorithms; avoid upstream's unrelated OMPL planners and FetchContent copy.
add_library(dapeng_wavestar STATIC
  ${WAVESTAR_SOURCE_DIR}/src/a_star.cc
  ${WAVESTAR_SOURCE_DIR}/src/theta_star.cc
  ${WAVESTAR_SOURCE_DIR}/src/multi_res_theta_star.cc
  ${WAVESTAR_SOURCE_DIR}/src/planner_base.cc
  ${WAVESTAR_SOURCE_DIR}/src/pointwise_heuristics.cc
  ${WAVESTAR_SOURCE_DIR}/src/volumetric_heuristics.cc)
target_include_directories(dapeng_wavestar PUBLIC ${WAVESTAR_SOURCE_DIR}/include)
target_link_libraries(dapeng_wavestar PUBLIC wavemap_core)
add_library(dapeng_navigation_core src/map_manager.cpp src/planning_map.cpp src/planner.cpp src/recorder.cpp src/visualization.cpp src/surface_mesh.cpp)
target_include_directories(dapeng_navigation_core PUBLIC include)
target_link_libraries(dapeng_navigation_core PUBLIC dapeng_wavestar wavemap_core wavemap_pipeline wavemap_io yaml-cpp Threads::Threads)
target_compile_options(dapeng_navigation_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(offline_surface_check src/offline_surface_check.cpp)
target_link_libraries(offline_surface_check PRIVATE dapeng_navigation_core)
add_executable(offline_check src/offline_check.cpp)
target_link_libraries(offline_check PRIVATE dapeng_navigation_core)
option(DAPENG_NAVIGATION_TESTS "Build independent navigation acceptance tests" ON)
if(DAPENG_NAVIGATION_TESTS)
  enable_testing()
  add_executable(fixture_map test/fixture_map.cpp)
  target_link_libraries(fixture_map PRIVATE dapeng_navigation_core)
  add_executable(test_map_management test/test_map_management.cpp)
  target_link_libraries(test_map_management PRIVATE dapeng_navigation_core)
  add_test(NAME map_operations_and_rolling_map COMMAND test_map_management)
  add_executable(test_planning_map test/test_planning_map.cpp)
  target_link_libraries(test_planning_map PRIVATE dapeng_navigation_core)
  add_test(NAME sdf_classification_and_versions COMMAND test_planning_map)
  add_executable(test_planner test/test_planner.cpp)
  target_link_libraries(test_planner PRIVATE dapeng_navigation_core)
  add_test(NAME wavestar_algorithms_and_collisions COMMAND test_planner)
  add_executable(test_recorder test/test_recorder.cpp)
  target_link_libraries(test_recorder PRIVATE dapeng_navigation_core)
  add_test(NAME bounded_metadata_recording COMMAND test_recorder)
  add_executable(test_surface_mesh test/test_surface_mesh.cpp)
  target_link_libraries(test_surface_mesh PRIVATE dapeng_navigation_core)
  add_test(NAME incremental_surface_geometry COMMAND test_surface_mesh)
  add_executable(test_visualization test/test_visualization.cpp)
  target_link_libraries(test_visualization PRIVATE dapeng_navigation_core)
  add_test(NAME native_incremental_visualization COMMAND test_visualization)
endif()
